sudo tee /var/ossec/integrations/custom-n8n-genai >/dev/null <<'PY'
#!/usr/bin/env python3
import json
import sys
import urllib.request
import urllib.error

def fail(msg):
    print(f"custom-n8n-genai error: {msg}", file=sys.stderr)
    sys.exit(1)

if len(sys.argv) < 4:
    fail("usage: custom-n8n-genai <alert_file> <api_key> <hook_url>")

alert_file = sys.argv[1]
api_key = sys.argv[2] if len(sys.argv) > 2 else ""
hook_url = sys.argv[3] if len(sys.argv) > 3 else ""

if not hook_url:
    fail("missing hook_url")

try:
    with open(alert_file, "r", encoding="utf-8") as f:
        alert = json.load(f)
except Exception as e:
    fail(f"could not read alert file: {e}")

rule_id = str(alert.get("rule", {}).get("id", ""))
groups = alert.get("rule", {}).get("groups", []) or []

target_rule_ids = {"100201", "100202", "100203"}
target_groups = {
    "genai_prompt_injection",
    "genai_indirect_injection",
    "genai_output_handling",
}

if rule_id not in target_rule_ids and not any(g in target_groups for g in groups):
    print(f"custom-n8n-genai ignored non-target alert rule_id={rule_id}")
    sys.exit(0)

payload = json.dumps(alert).encode("utf-8")

headers = {
    "Content-Type": "application/json",
    "User-Agent": "wazuh-custom-n8n-genai/1.0",
}

if api_key and api_key not in {"none", "NONE", "-"}:
    headers["Authorization"] = f"Bearer {api_key}"

req = urllib.request.Request(
    hook_url,
    data=payload,
    headers=headers,
    method="POST",
)

try:
    with urllib.request.urlopen(req, timeout=10) as resp:
        body = resp.read().decode("utf-8", errors="replace")
        print(f"custom-n8n-genai delivered rule_id={rule_id} status={resp.status} body={body[:200]}")
except urllib.error.HTTPError as e:
    body = e.read().decode("utf-8", errors="replace")
    fail(f"http error {e.code}: {body[:500]}")
except Exception as e:
    fail(f"delivery failed: {e}")
PY

sudo chmod 750 /var/ossec/integrations/custom-n8n-genai
